  # ── Test records ──────────────────────────────────────────────────────────
  # `task test` and `test:integration` record
  # what they ran into .forge/tests/runs/<id>/: `forge test begin` allocates
  # the run (or joins $FORGE_TEST_RUN, which `forge test run` sets so several
  # lanes form one run), and `forge test capture` runs the lane's command
  # UNCHANGED, stores its results and coverage, prints ordinary `go test`
  # output and exits with the command's own status. `forge test report`
  # (`task test:report`) summarises the record. `test:short` records nothing.
  #
  # Recording never decides a verdict, and never breaks a lane: with
  # FORGE_TESTS=off, or a forge that cannot begin a run (one that predates
  # `forge test`, an offline `go run`), RUN is empty and every lane runs
  # exactly the command it would without recording — no capture, no -json,
  # no coverage flags.
  #
  # TEST_BEGIN prints the run directory, or nothing. Each lane evaluates it
  # in its own vars, so a `task` that runs no test lane never pays for it.
  TEST_BEGIN: '[ "$FORGE_TESTS" = off ] || { d="$({{.FORGE}} test begin)" && [ -d "$d" ] && echo "$d"; } || true'
  # TEST_SKIP is the `go test -skip` pattern of the quarantined tests
  # (ci/quarantine/go/, written by `forge test quarantine add`), or nothing.
  # forge is asked only when something is quarantined.
  TEST_SKIP: '{ [ ! -d ci/quarantine/go ] && [ -z "$CI_QUARANTINE_EXTRA" ]; } || {{.FORGE}} test quarantine skip-regex go || true'
  # COVERAGE is forge.yaml's `tests.coverage`: `true`/`false`, or a map whose
  # `enabled` defaults to true (the legacy `ci.test.coverage` when the tests
  # block does not say), always rendered as exactly "true" or "false". "false"
  # keeps the coverage flags out of every lane, which then record results only.
  # COVERPKG is ` -coverpkg=./...` under `tests.coverage.packages: module`:
  # each test binary then counts the whole module, not just its own package.
  COVERAGE: >-
    {{- $cfg := fromYaml .FORGE_YAML -}}
    {{- $tests := index $cfg "tests" -}}
    {{- $cov := "" -}}
    {{- if kindIs "map" $tests}}{{$cov = index $tests "coverage"}}{{end -}}
    {{- if kindIs "bool" $cov}}{{if $cov}}true{{else}}false{{end}}
    {{- else if kindIs "map" $cov}}{{if hasKey $cov "enabled"}}{{eq (toString (index $cov "enabled")) "true"}}{{else}}true{{end}}
    {{- else}}{{$ci := index $cfg "ci"}}{{$t := ""}}{{if kindIs "map" $ci}}{{$t = index $ci "test"}}{{end}}{{if kindIs "map" $t}}{{eq (toString (index $t "coverage")) "true"}}{{else}}false{{end}}
    {{- end -}}
  COVERPKG: >-
    {{- $tests := index (fromYaml .FORGE_YAML) "tests" -}}
    {{- $cov := "" -}}
    {{- if kindIs "map" $tests}}{{$cov = index $tests "coverage"}}{{end -}}
    {{- if and (kindIs "map" $cov) (eq (toString (index $cov "packages")) "module")}} -coverpkg=./...{{end -}}
